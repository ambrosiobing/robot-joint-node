  {the bus controller, chapters 9 to 13};

\draw[flow] (ctl.north -| nfr) -- (nfr.south);

% ================================================================ the decision
\node[regcell, fill=hwcol, minimum width=44mm, text width=42mm, inner sep=3pt,
      align=left, anchor=north west] at (0.4,2.7)
  {\textbf{framing\_enabled = false}\\[2pt]
   A bus frame already carries its own length, so the framing a byte stream
   needs is exactly what must not be applied here.};
\node[regcell, fill=usrcol, minimum width=44mm, text width=42mm, inner sep=3pt,
      align=left, anchor=north west] at (0.4,0.6)
  {\textbf{63 bytes, fixed}\\[2pt]
   The transport unit is set in the client's own header. Anything larger fails
   without an error message that says so.};

% ================================================================ the host side
\node[layerlabel] at (0.4,-1.6) {on the host};
\node[ext, text width=44mm, minimum height=9mm] (cagent) at (2.8,-2.35)
  {the custom agent class, with the same identifier};
\node[ext, text width=44mm, minimum height=9mm] (sock) at (2.8,-3.55)
  {the host's own bus interface, which must carry 64-byte payloads};
\draw[flow] (cagent) -- (sock);
\draw[bus] (ctl.east) -- (sock.west);
\node[buslabel] at (-0.15,-2.6) {the bus};

\node[umlnote, text width=52mm, anchor=north west] at (-6.6,-4.4)
  {The agent ships a bus transport of its own, and it is the reference for what
   the far side has to do. What it does not ship, and what the client does not
   ship either, is the node-side implementation for a small kernel: the shipped
   one is for a full operating system and its socket layer.};
\node[umlnote, text width=52mm, anchor=north west] at (0.4,-4.4)
  {These six functions are the whole port. They are short, they are the only
   place in the node that knows both the middleware and the controller, and they
   are the one piece of this chapter that does not exist anywhere else for this
   kernel.};
