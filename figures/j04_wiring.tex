% The pin budget for the whole volume. This is a logical map, not a drawing of
% the connectors: the physical positions are in the board manual and are
% confirmed there, row by row, before any chapter claims a pin.
\begin{tikzpicture}[font=\sffamily\small, x=1cm, y=1cm]

\tikzset{
  pinused/.style={draw=linecol, fill=usrcol, minimum width=16mm,
                  minimum height=5.5mm, font=\sffamily\scriptsize, align=center},
  pinres/.style={pinused, fill=hwcol},
  pinfree/.style={pinused, fill=black!5, text=black!55},
}

% ---------------- claimed and in use
\node[chlabel, anchor=west] at (-0.2,4.55) {claimed and in use};
\foreach \i/\p/\f in {0/PB0/{LED green}, 1/PE1/{LED yellow}, 2/PB14/{LED red},
                      3/PC13/{button}, 4/PD8/{console Tx}, 5/PD9/{console Rx},
                      6/PA0/{event in}}{
  \node[pinused] at (\i*1.75,4.0) {\p\\{\tiny \f}};
}
\node[chlabel, anchor=west] at (12.3,4.0) {chapters 1 to 3};

% ---------------- claimed and reserved
\node[chlabel, anchor=west] at (-0.2,2.85) {claimed and reserved};
\foreach \i/\p/\f in {0/PD1/{bus Tx}, 1/PD0/{bus Rx}, 2/{TIM1}/{PWM, 3 pairs},
                      3/{TIM1 BKIN}/{fault in}, 4/{TIM x}/{encoder A, B},
                      5/{EXTI}/{index}, 6/{I2C}/{shield}}{
  \node[pinres] at (\i*1.75,2.3) {\p\\{\tiny \f}};
}
\node[chlabel, anchor=west] at (12.3,2.3) {chapters 5 to 13};

% ---------------- the two that are not yet confirmed
\draw[faultedge] (0,1.85) -- (0,1.3);
\draw[faultedge] (1.75,1.85) -- (1.75,1.3);
\node[umlnote, text width=52mm, anchor=north west] at (-0.9,1.25)
  {These two rows carry the marker \texttt{TO CONFIRM} in the board description,
   because the pin pair for the bus peripheral has not yet been checked against
   the board manual. The generator refuses to emit code for a row in that state,
   so the question becomes a build error in chapter 9 rather than a silent
   assumption today.};

% ---------------- what is left
\node[chlabel, anchor=west] at (6.6,0.55) {everything else on the two headers};

  \node[pinfree] at (6.6+\i*1.75,0.0) {free};
}
\node[chlabel, anchor=west] at (13.3,0.0) {and a great many more};

\node[umlnote, text width=48mm, anchor=north west] at (6.4,-0.65)
  {A reader who wants to add something to this node reads this figure first. A
   chapter that takes a free pin moves it up a row here, and a chapter that
   discovers a collision with a fitted shield says so in its own text rather
   than quietly choosing another pin.};

\node[chlabel, anchor=north west, text width=58mm] at (-0.9,-1.9)
  {Physical positions, the shield rows and the jumper options live in the board
