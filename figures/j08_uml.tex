\newcommand{\umlbox}[4]{%
  \node[umlclass, text width=36mm] (#1) at (#2) {%
    \begin{minipage}{36mm}
      \centering\bfseries #3\par\vspace{1pt}
      \hrule\vspace{2pt}
      \raggedright\normalfont #4\vspace{1pt}
    \end{minipage}};}

% ---------------- the real inputs
\umlbox{pos}{-5.0,3.2}{position}{%
  from the real decode path\\ chapter 5\\[1pt]
  {\itshape the only real input}}
\umlbox{cmd}{0,3.2}{commanded torque}{%
  k\_t times duty, chapter 7\\[1pt]
  {\itshape modelled: there is no\\ current loop}}
\umlbox{par}{5.0,3.2}{plant parameters}{%
  J, b, tau\_c\\[1pt]
  {\itshape chosen, never measured}}

% ---------------- the observer
\umlbox{diff}{-5.0,0.6}{second difference}{%
  then a filter\\[1pt]
  {\itshape its latency is the\\ detector's response time}}
\umlbox{obs}{0,0.6}{observer}{%
  J*acc - tau\_cmd + friction\\[1pt]
  {\itshape every model error\\ appears in this number}}
\umlbox{prov}{5.0,0.6}{provenance}{%
  value, source, model version,\\ filter latency\\[1pt]
  {\itshape travels with the number}}

% ---------------- the decision
\umlbox{det}{0,-2.0}{contact detector}{%
  threshold, hold, latch\\[1pt]
  {\itshape tuned by a measured\\ false-alarm rate}}

% ---------------- the one real measurement
\umlbox{imu}{5.0,-2.0}{the inertial unit}{%
  chapter 6, real silicon\\[1pt]
  {\itshape sees the bench move.\\ A different question}}

\draw[flow] (pos) -- (diff);
\draw[flow] (diff) -- (obs);
\draw[flow] (cmd) -- (obs);
\draw[flow] (par) -- (obs);
\draw[flow] (obs) -- (prov);
\draw[flow] (obs) -- (det);
\draw[flow] (imu) -- node[lbl, align=center]{compared,\\never fused} (det);

\node[umlnote, text width=52mm, anchor=north west] at (-7.4,-3.4)
  {The estimate is reported and is not fed back into the controller in this
   volume. An estimate whose error contains the model's own mistakes, used as a
   control input, closes a loop through those mistakes. Chapter 16 controls on
   position; chapter 18 may act on this detector's decision.};
\node[umlnote, text width=52mm, anchor=north west] at (0.4,-3.4)
  {The two detectors are compared and never fused. They answer different
   questions: the observer sees a torque the model cannot explain, and the
   inertial unit sees the bench move. Fusing them would produce one number that
   answers neither question cleanly.};
\end{tikzpicture}
