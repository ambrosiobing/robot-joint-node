\node[regcell, fill=red!18, minimum width=54mm, text width=52mm, inner sep=3pt,

   clears because the condition went away, and a lifetime count survives a
   reset.\\[2pt]
   \textbf{No citable source states this, or that the safe state must be
   reachable from everywhere.} Both are practice, and this volume says so rather
   than attributing them.};

\node[regcell, fill=usrcol, minimum width=54mm, text width=52mm, inner sep=3pt,

   action; available in every mode; overrides everything without impairing other
   protections; \textbf{maintained until manually reset}; no start command
   effective meanwhile; and \textbf{the reset itself shall not initiate a
   restart}.};

\node[umlnote, text width=114mm, anchor=north west] at (-6.8,-4.1)

   function, and nothing more, and writing that down is what makes the rest of
   this chapter credible.};
