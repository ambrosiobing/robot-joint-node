% One period, from the update event to the histogram, and the report path that
% is deliberately outside the handler. Vertical spacing is generous on purpose:
% a sequence diagram whose labels touch is a sequence diagram nobody reads.
\begin{tikzpicture}[font=\sffamily\scriptsize, x=1cm, y=1cm]

\foreach \x/\name/\id in {0/TIM6/tim, 2.9/{period handler}/isr, 5.8/{cycle counter}/dwt,
                          8.7/histogram/hist, 11.6/{idle loop}/idle, 14.5/console/con}{
  \node[actor, text width=19mm] (\id) at (\x,0) {\name};
  \draw[lifeline] (\id.south) -- ++(0,-10.2);
}

\draw[msg] (0,-1.1) -- node[above]{update event, every 280,000 cycles} (2.9,-1.1);
\node[activation, minimum height=6.6mm] at (2.9,-1.45) {};

\draw[msg]   (2.9,-2.1) -- node[above]{read} (5.8,-2.1);
\draw[reply] (5.8,-2.9) -- node[below]{now} (2.9,-2.9);

\draw[msg] (2.9,-3.7) -- ++(1.0,0) -- ++(0,-0.5)
  node[right, xshift=1mm]{delta = now - last, unsigned} -- ++(-1.0,0);

\draw[msg] (2.9,-5.2) -- node[above]{record delta, min, max, seq} (8.7,-5.2);
\node[activation, minimum height=3.2mm] at (8.7,-5.45) {};

  node[right, xshift=1mm]{loop\_step(), timed separately} -- ++(-1.0,0);

\node[umlnote, text width=30mm, anchor=north east] at (-0.3,-1.6)
  {The handler does the minimum: no printing, no allocation, no lock that
   anything else holds. A polled console write at 115200 takes longer than the
   period it would be reporting.};
\draw[dotted, draw=linecol] (-0.3,-2.9) -- (2.6,-2.3);

\node[umlnote, text width=30mm, anchor=north east] at (-0.3,-6.2)
  {The report reads while the handler writes. The fix is a sequence counter
   rather than a lock: a lock would add latency to the handler, and a statistic
   can afford one retry.};
\draw[dotted, draw=linecol] (-0.3,-7.6) -- (8.4,-9.0);
\end{tikzpicture}
