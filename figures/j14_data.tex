% What the middleware costs: the published figure, the caps, and the refusal.
\begin{tikzpicture}[font=\sffamily\scriptsize, x=1cm, y=1cm]

\node[chlabel, anchor=west] at (-6.6,3.3) {a footprint has three parts, and only one of them is published};

% ================================================================ the published figure
\node[regcell, fill=usrcol, minimum width=52mm, text width=50mm,
      anchor=north west, align=left, inner sep=3pt] at (-6.6,2.8)
  {\textbf{1. the published figure}\\[2pt]
   Under 75 kB of flash and around 3 kB of memory, for a complete publisher and
   subscriber application with messages of about 512 bytes.\\[2pt]
   \textbf{Its scope is the constrained-environments client alone}, not the four
   layers above it.};

% ================================================================ the task stack
\node[regcell, fill=kercol, minimum width=52mm, text width=50mm,
      anchor=north west, align=left, inner sep=3pt] at (-6.6,-0.25)
  {\textbf{3. the task stack}\\[2pt]
   More than 10 kB, which the integration's own notes state as a requirement.
   Sized here at 12 kB, and the symptom of getting it wrong is a fault rather
   than a message.};

% ================================================================ the refusal
\node[regcell, fill=red!18, minimum width=52mm, text width=50mm,
      anchor=north west, align=left, inner sep=3pt] at (-6.6,-2.65)
  {\textbf{and what this volume will not print}\\[2pt]
   The widely repeated minimum of 32 kB of memory and 256 kB of flash. It comes
   from a commercial blog, not from the project, and no official figure in text
   form for the complete stack on this core was found at all.};

% ================================================================ the caps
\node[regcell, fill=white, minimum width=52mm,
      anchor=north west, align=left, inner sep=3pt] at (-0.6,2.8)
  {\begin{tabular}{@{}lccl@{}}
   \multicolumn{4}{@{}l@{}}{\textbf{2. the caps, which are the rest of the cost}}\\[3pt]
   cap & default & here & why \\[1pt]
   \hline
   nodes & 1 & 1 & one joint \\
   publishers & 4 & 2 & state, diagnostics \\
   subscriptions & 4 & 1 & configuration \\
   services & 4 & 1 & one request path \\
   clients & 4 & 0 & it calls nobody \\
   history & 8 & 4 & nothing replays \\
   stream history & 4 & 4 & a power of two \\
   graph support & off & off & the agent knows \\
   dynamic alloc & off & off & and stays off \\
   \end{tabular}};

\node[chlabel, anchor=north west, text width=52mm] at (-0.6,-1.55)
  {Every reduction carries its reason in the file, because a cap trimmed without
   one is the first thing somebody restores when a build fails. Left alone, the
   defaults are four of each, a history of eight, and both optional features
   already off.};

\node[umlnote, text width=112mm, anchor=north west] at (-6.6,-5.4)
  {Chapter 1 budgeted the whole node at 512 kB of flash and 256 kB of memory, so
   that a number here would mean something. That is the denominator this chapter
   reports against. A middleware that takes a fifth of a node's entire flash
   budget is a legitimate engineering decision and an illegitimate surprise, and
   the difference between the two is whether the figure was stated before the
   build or discovered after it.};
\end{tikzpicture}
