% ================================================================ the bit
\draw[faultedge, -] (60,-7) rectangle (74,20);
\node[tlabel, anchor=south, text=red!60!black] at (67,20.5) {this one bit};
\node[chlabel, anchor=north west, text width=60mm] at (46,-8)
  {dominant in the older format, recessive in the newer one, and in exactly the same position in
   both. A controller that predates the newer format reads it as a malformed frame};

% ================================================================ what happens next
\node[chlabel, anchor=west] at (0,-20) {and what a controller without tolerance puts on the wire next};
\begin{scope}[yshift=-30mm]
  \framefield{0}{61}{fstuff}{the frame so far, already transmitted}
  \framefield{61}{34}{fieldbox, fill=red!25}{an error flag: dominant bits}
  \framefield{95}{24}{fstuff}{error delimiter}
\end{scope}
\node[tlabel, anchor=north west, text width=56mm] at (0,-32)
  {The flag is deliberately a run of identical bits, which breaks the stuffing rule, so every
   participant on the bus detects it whether or not it understood the frame.};

% ================================================================ notes
\node[umlnote, text width=52mm, anchor=north west] at (60,-32)
  {A controller documented as tolerant reads the same bit, concludes that the frame is in a format
   it cannot decode, and stays quiet until the bus is idle again. The difference between the two
   behaviours is one line in a datasheet and the whole usability of the bus.};
\end{tikzpicture}
